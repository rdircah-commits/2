document.getElementById("languageSelect").addEventListener("change", function () {
    currentLanguage = this.value;
    applyLanguage();

function applyLanguage() {

    const t = translations[currentLanguage];

    document.querySelectorAll("[data-i18n]").forEach(el => {
        const key = el.dataset.i18n;

        if (t[key]) {
            el.textContent = t[key];
        }
    });

    document.querySelectorAll("[data-placeholder]").forEach(el => {
        const key = el.dataset.placeholder;

        if (t[key]) {
            el.placeholder = t[key];
        }
    });

    document.documentElement.lang = currentLanguage;

    showToast(
        currentLanguage === "zh-TW"
        ? "語言已切換"
        : "Language changed"
    );
}

function goTo(id) {

    const target = document.getElementById(id);

    if (!target) return;

    target.scrollIntoView({
        behavior: "smooth",
        block: "start"
    });

    document.querySelectorAll(".side-link").forEach(btn => {
        btn.classList.remove("active");
    });

    event.currentTarget.classList.add("active");
}

/* =========================================================
   COLLAPSE
========================================================= */

function collapse(button) {

    const section = button.closest(".section");
    const body = section.querySelector(".section-body");

    if (body.classList.contains("hidden")) {
        body.classList.remove("hidden");
        button.textContent = "⌃";
    } else {
        body.classList.add("hidden");
        button.textContent = "⌄";
    }
}

function toggleAll() {

    const bodies = document.querySelectorAll(".section-body");

    const shouldOpen =
        [...bodies].some(body => body.classList.contains("hidden"));

    bodies.forEach(body => {

        body.classList.toggle("hidden", !shouldOpen);

        const btn =
            body.closest(".section")
                .querySelector(".section-head .icon-btn");

        if (btn) {
            btn.textContent = shouldOpen ? "⌃" : "⌄";
        }
    });
}

        .value

        .toLowerCase();

    document.querySelectorAll(".searchable").forEach(section => {

        const text = section.innerText.toLowerCase();

        if (!keyword || text.includes(keyword)) {
            section.classList.remove("hidden");
        } else {
            section.classList.add("hidden");
        }
    });
}

        "pinkcare-dark",

/* =========================================================
   SAVE
========================================================= */

function collectFormData() {

    const data = {};

    document.querySelectorAll("input[name], textarea[name]").forEach(el => {
        data[el.name] = el.value;
    });

    data.language = currentLanguage;
    data.updatedAt = new Date().toISOString();

    return data;
}

function saveRecord() {

    const data = collectFormData();

    localStorage.setItem(
        "pinkcare-emr",
        JSON.stringify(data)
    );

    showToast(
        currentLanguage === "zh-TW"
        ? "病歷已保存到本機瀏覽器"
        : "Medical record saved locally"
    );
}

/* =========================================================
   LOAD
========================================================= */

function loadRecord() {

    const saved = localStorage.getItem("pinkcare-emr");

    if (!saved) return;

    try {

        const data = JSON.parse(saved);

        Object.keys(data).forEach(key => {

            const el =
                document.querySelector(`[name="${key}"]`);

            if (el) {
                el.value = data[key];
            }
        });

    } catch (error) {
        console.warn("Could not load local record.");
    }
}

/* =========================================================
   EXPORT JSON
========================================================= */

function exportJSON() {

    const data = collectFormData();

    const blob = new Blob(
        [JSON.stringify(data, null, 2)],
        {type: "application/json"}
    );

    const url = URL.createObjectURL(blob);

    const a = document.createElement("a");

    a.href = url;
    a.download =
        "pinkcare-medical-record-" +
        new Date().toISOString().slice(0,10) +
        ".json";

    a.click();

    URL.revokeObjectURL(url);

    showToast("JSON exported");
}

/* =========================================================
   IMPORT JSON
========================================================= */

function importJSON(event) {

    const file = event.target.files[0];

    if (!file) return;

    const reader = new FileReader();

    reader.onload = function(e) {

        try {

            const data = JSON.parse(e.target.result);

            Object.keys(data).forEach(key => {

                const el =
                    document.querySelector(`[name="${key}"]`);

                if (el) {
                    el.value = data[key];
                }
            });

            saveRecord();

            showToast("JSON imported");

        } catch (error) {

            alert("JSON 格式無法讀取。");

        }
    };

    reader.readAsText(file);
}

/* =========================================================
   VOICE INPUT
========================================================= */

let activeRecognition = null;

function startVoice(button) {

    const SpeechRecognition =
        window.SpeechRecognition ||
        window.webkitSpeechRecognition;

    if (!SpeechRecognition) {

        alert(
            "目前瀏覽器不支援 Web Speech API。\n" +
            "建議使用最新版 Chrome 或 Edge。"
        );

        return;
    }

    const textarea =
        button.parentElement.querySelector("textarea");

    if (!textarea) return;

    if (activeRecognition) {

        activeRecognition.stop();
        activeRecognition = null;

        button.classList.remove("recording");

        return;
    }

    const recognition =
        new SpeechRecognition();

    recognition.lang =
        getSpeechLanguage(currentLanguage);

    recognition.continuous = false;
    recognition.interimResults = true;

    activeRecognition = recognition;

    button.classList.add("recording");

    const original =
        textarea.value
        ? textarea.value + " "
        : "";

    recognition.onresult = function(event) {

        let transcript = "";

        for (
            let i = event.resultIndex;
            i < event.results.length;
            i++
        ) {
            transcript +=
                event.results[i][0].transcript;
        }

        textarea.value =
            original + transcript;
    };

    recognition.onerror = function(event) {

        console.warn("Speech recognition:", event.error);

        button.classList.remove("recording");
        activeRecognition = null;
    };

    recognition.onend = function() {

        button.classList.remove("recording");
        activeRecognition = null;
    };

    recognition.start();
}

function getSpeechLanguage(lang) {

    const map = {
        "zh-TW": "zh-TW",
        "zh-CN": "zh-CN",
        "en": "en-US",
        "ja": "ja-JP",
        "ko": "ko-KR",
        "th": "th-TH",
        "la": "la"
    };

    return map[lang] || "zh-TW";
}

/* =========================================================
   TEXT TO SPEECH
========================================================= */

function speakRecord() {

    if (!("speechSynthesis" in window)) {

        alert("此瀏覽器不支援語音朗讀。");

        return;
    }

    speechSynthesis.cancel();

    const data = collectFormData();

    const text = Object.entries(data)
        .filter(([key]) => key !== "language" && key !== "updatedAt")
        .map(([key, value]) => `${key}: ${value}`)
        .join(". ");

    const utterance =
        new SpeechSynthesisUtterance(text);

    utterance.lang =
        getSpeechLanguage(currentLanguage);

    utterance.rate = 0.9;

    speechSynthesis.speak(utterance);
}
